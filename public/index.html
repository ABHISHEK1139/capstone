Python Estimator Quickstart

Drop-in compatibility with scikit-learn pipelines and sparse transaction matrices.

from sklearn.pipeline import Pipeline
from skarm import TransactionEncoder, AssociationRulesEstimator

pipe = Pipeline([
    ("encode", TransactionEncoder(sparse=True)),
    ("mine", AssociationRulesEstimator(
        algorithm="auto",       # "apriori", "fpgrowth", or "eclat"
        min_support=0.02,
        metric="lift",
        min_threshold=1.2,
        max_candidates=100_000,
    )),
])

pipe.fit(transactions)
rules = pipe.named_steps["mine"].get_rules()
top = pipe.named_steps["mine"].explore().filter_rules(min_lift=1.5, sort_by="confidence")
